# gam-mcp 0.6.0 PR 1 — One Report Path, Typed Results, Pages

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Every report goes through one function that returns typed rows, GAM's own provenance and a stateless cursor for further pages; the seven polling loops, the CSV round trip and `file_path` output are gone.

**Architecture:** New modules `values.ts` (GAM value → typed cell), `report-definition.ts` (the JSON sent to GAM), `envelope.ts` (result shape with qualifiers before rows), `cursor.ts` (page cursor) and `report-exec.ts` (define → run → poll → fetch). The GAM client gains `defineReport`, `runReport`, `fetchRows` and `getReport`; tools move onto `report-exec.ts` in groups, and the old `createReport`/`downloadReport`/`output.ts` path is deleted last. Each task leaves the suite green.

**Tech Stack:** TypeScript (strict, ESM/nodenext), Node ≥ 22, zod 3, `@modelcontextprotocol/sdk` 1.x, vitest 4.

**Spec:** `docs/superpowers/specs/2026-09-17-gam-mcp-0.6.0-design.md` (§1, §2.1, §1.5). Index: `docs/superpowers/plans/2026-09-17-gam-mcp-0.6.0-index.md`.

## Global Constraints

- Node ≥ 22, TypeScript strict, ESM; relative imports end in `.js`.
- No new runtime dependencies. SDK stays v1.
- `npm run typecheck` covers `src/` only. **Tests are not typechecked** — always run them.
- Conventional Commits; **no `Co-Authored-By` or AI attribution**.
- Tests use network code `1234` and invented names only (generic-MCP rule).
- Result key order is a contract: `summary`, `provenance`, `total_row_count`, `returned_row_count`, `truncated`, `next_cursor`, `warnings`, `schema`, `rows` (spec §1.1).
- `GAM_PAGE_ROWS` default 5,000; GAM's `pageSize` maximum is 10,000 (spec §1.3).
- Tools that compute from rows read at most 50,000 input rows (spec §1.4).
- Out of scope for this PR (do not implement): report reuse, `displayName`, retries policy, network identity, time zone/currency provenance, the result cache, strict inputs, date validation, annotations, `derived` blocks, `null` arithmetic. See the index for where each lands.

```ts
import { fieldType } from './schema.js';

// GAM's ReportValue union (Ad Manager API v1). int64 values arrive as JSON
// strings; proto3 JSON omits unset fields, so an empty object means "no value".
export type ReportValue = {
  intValue?: string;
  doubleValue?: number;
  stringValue?: string;
  boolValue?: boolean;
  bytesValue?: string;
  intListValue?: { values?: string[] };
  doubleListValue?: { values?: number[] };
  stringListValue?: { values?: string[] };
};

export type ReportRow = {
  dimensionValues?: ReportValue[];
  metricValueGroups?: { primaryValues?: ReportValue[] }[];
};

export type Scalar = string | number | boolean;
export type Cell = Scalar | Scalar[] | null;
export type ColumnType = 'integer' | 'number' | 'string' | 'boolean' | 'date' | 'array' | 'bytes';
export type Column = { name: string; type: ColumnType };
export type Table = { columns: Column[]; rows: Record<string, Cell>[]; warnings: string[] };

type Converted = { value: Cell; type: ColumnType | null; unsafeInteger: boolean };

const NONE: Converted = { value: null, type: null, unsafeInteger: false };
const YYYYMMDD = /^\d{8}$/;

function integer(s: string): { value: number | string; unsafe: boolean } {
  const n = Number(s);
  return Number.isSafeInteger(n) ? { value: n, unsafe: false } : { value: s, unsafe: true };
}

export function convertValue(v: ReportValue | null | undefined, isDateColumn: boolean): Converted {
  if (!v) return NONE;
  if (v.intValue !== undefined) {
    if (isDateColumn && YYYYMMDD.test(v.intValue)) {
      const s = v.intValue;
      return { value: `${s.slice(0, 4)}-${s.slice(4, 6)}-${s.slice(6, 8)}`, type: 'date', unsafeInteger: false };
    }
    const { value, unsafe } = integer(v.intValue);
    return { value, type: 'integer', unsafeInteger: unsafe };
  }
  if (v.doubleValue !== undefined) {
    // proto3 JSON writes NaN/Infinity as strings; they are not numbers a reader can use.
    return typeof v.doubleValue === 'number' && Number.isFinite(v.doubleValue)
      ? { value: v.doubleValue, type: 'number', unsafeInteger: false }
      : NONE;
  }
  if (v.stringValue !== undefined) {
    return { value: v.stringValue, type: isDateColumn ? 'date' : 'string', unsafeInteger: false };
  }
  if (v.boolValue !== undefined) return { value: v.boolValue, type: 'boolean', unsafeInteger: false };
  if (v.bytesValue !== undefined) return { value: v.bytesValue, type: 'bytes', unsafeInteger: false };
  if (v.intListValue) {
    const parts = (v.intListValue.values ?? []).map(integer);
    return { value: parts.map((p) => p.value), type: 'array', unsafeInteger: parts.some((p) => p.unsafe) };
  }
  if (v.doubleListValue) return { value: [...(v.doubleListValue.values ?? [])], type: 'array', unsafeInteger: false };
  if (v.stringListValue) return { value: [...(v.stringListValue.values ?? [])], type: 'array', unsafeInteger: false };
  return NONE;
}

const FALLBACK: Record<string, ColumnType> = {
  string: 'string',
  number: 'number',
  date: 'date',
  currency: 'number',
  rate: 'number',
};

// Columns are named from the report definition (dimensions, then metrics),
// never from process memory, so a restarted server still labels them.
export function buildTable(dimensions: string[], metrics: string[], rows: ReportRow[]): Table {
  const names = [...dimensions, ...metrics];
  const dateColumn = names.map((n) => fieldType(n) === 'date');
  const types: (ColumnType | null)[] = names.map(() => null);
  const unsafe = new Set<string>();

  const out = rows.map((row) => {
    const values = [
      ...(row.dimensionValues ?? []),
      ...(row.metricValueGroups?.[0]?.primaryValues ?? []),
    ];
    const record: Record<string, Cell> = {};
    names.forEach((name, i) => {
      const c = convertValue(values[i], dateColumn[i]);
      record[name] = c.value;
      if (c.type && !types[i]) types[i] = c.type;
      if (c.unsafeInteger) unsafe.add(name);
    });
    return record;
  });

  const columns = names.map((name, i) => ({
    name,
    type: types[i] ?? FALLBACK[fieldType(name) ?? 'string'] ?? 'string',
  }));
  const warnings = [...unsafe].map(
    (n) => `Column ${n} holds integers beyond ±2^53; those values are strings so they stay exact.`,
  );
  return { columns, rows: out, warnings };
}

// Text form of a cell, for map keys and labels. A missing value is the empty
// string here only because keys need a string; arithmetic must not use this.
export function cellText(c: Cell | undefined): string {
  if (c === null || c === undefined) return '';
  return Array.isArray(c) ? c.join(',') : String(c);
}
```

```ts
import type { DateRange, ReportQuery } from './gam-client.js';

export type GamDate = { year: number; month: number; day: number };

// The reportDefinition object GAM receives, in GAM's own shape. Provenance
// copies filters and sorts from here, so what a result claims is what ran.
export type ReportDefinition = {
  dimensions: string[];
  metrics: string[];
  dateRange: { fixed: { startDate: GamDate; endDate: GamDate } };
  reportType: string;
  filters?: unknown[];
  customDimensionKeyIds?: string[];
};

function toDateParts(iso: string): GamDate {
  const [year, month, day] = iso.split('-').map(Number);
  return { year, month, day };
}

export function reportDefinition(query: ReportQuery, dateRange: DateRange): ReportDefinition {
  return {
    dimensions: query.dimensions,
    metrics: query.metrics,
    dateRange: {
      fixed: {
        startDate: toDateParts(dateRange.start),
        endDate: toDateParts(dateRange.end),
      },
    },
    reportType: query.report_type ?? 'HISTORICAL',
    ...(query.filters?.length && { filters: toReportFilters(query.filters) }),
    // GAM expects int64 IDs as strings. Index here = slot number on
    // CUSTOM_DIMENSION_<N>_VALUE dimensions.
    ...(query.custom_dimension_key_ids?.length && {
      customDimensionKeyIds: query.custom_dimension_key_ids.map(String),
    }),
  };
}
```

```ts
// In-process fake of the Ad Manager REST endpoints the client uses. Tests hand
// its `fetch` to createRestGamClient, so the real client code runs. Every
// request is logged; outcomes can be injected per request.
import type { ReportRow, ReportValue } from '../src/values.js';

const BASE = 'https://admanager.googleapis.com/v1/';

export const int = (n: number | string): ReportValue => ({ intValue: String(n) });
export const dbl = (n: number): ReportValue => ({ doubleValue: n });
export const str = (s: string): ReportValue => ({ stringValue: s });

type Json = Record<string, any>;
export type LoggedRequest = { method: string; path: string; body?: Json };
export type Outcome =
  | { status: number; body?: unknown }
  | 'network-error'
  // Perform the request, then answer with this status (a lost response).
  | { succeedThenStatus: number };

export type FakeGam = {
  fetch: typeof fetch;
  requests: LoggedRequest[];
  reports: Map<string, { name: string; displayName?: string; reportDefinition: Json }>;
  network: Json;
  setRows(rows: ReportRow[], extra?: { runTime?: string; dateRanges?: Json[] }): void;
  inject(method: string, path: RegExp, outcome: Outcome): void;
  expire(result: string): void;
};

export function createFakeGam(opts: { networkCode?: string; completeAfterPolls?: number } = {}): FakeGam {
  const code = opts.networkCode ?? '1234';
  const completeAfter = opts.completeAfterPolls ?? 1;
  const requests: LoggedRequest[] = [];
  const reports = new Map<string, { name: string; displayName?: string; reportDefinition: Json }>();
  const runs = new Map<string, { report: string; run: number; polls: number }>();
  const expired = new Set<string>();
  const injections: { method: string; path: RegExp; outcome: Outcome }[] = [];
  let rows: ReportRow[] = [];
  let extra: { runTime?: string; dateRanges?: Json[] } = {};
  let reportSeq = 0;
  let runSeq = 0;

  const network: Json = {
    name: `networks/${code}`,
    networkCode: code,
    displayName: 'Test network',
    timeZone: 'Asia/Tokyo',
    currencyCode: 'USD',
    secondaryCurrencyCodes: ['EUR'],
  };

  const json = (status: number, body: unknown) =>
    new Response(JSON.stringify(body), { status, headers: { 'content-type': 'application/json' } });
  const notFound = () => json(404, { error: { code: 404, status: 'NOT_FOUND' } });

  function route(method: string, path: string, body: Json | undefined): Response {
    const [pathname, query = ''] = path.split('?');
    const params = new URLSearchParams(query);
    let m: RegExpMatchArray | null;

    if (method === 'GET' && pathname === `networks/${code}`) return json(200, network);

    if (method === 'POST' && pathname === `networks/${code}/reports`) {
      const name = `networks/${code}/reports/${++reportSeq}`;
      const report = { name, ...(body?.displayName && { displayName: body.displayName }), reportDefinition: body?.reportDefinition ?? {} };
      reports.set(name, report);
      return json(200, report);
    }

    if (method === 'GET' && pathname === `networks/${code}/reports`) {
      const filter = params.get('filter');
      let list = [...reports.values()];
      if (filter !== null) {
        const f = /^displayName = "(.*)"$/.exec(filter);
        if (!f) return json(400, { error: { code: 400, status: 'INVALID_ARGUMENT', message: 'unsupported filter' } });
        list = list.filter((r) => r.displayName === f[1]);
      }
      return json(200, list.length ? { reports: list } : {});
    }

    if ((m = /^(networks\/\d+\/reports\/\d+)$/.exec(pathname)) && method === 'GET') {
      const r = reports.get(m[1]);
      return r ? json(200, r) : notFound();
    }

    if ((m = /^(networks\/\d+\/reports\/\d+):run$/.exec(pathname)) && method === 'POST') {
      if (!reports.has(m[1])) return notFound();
      const run = ++runSeq;
      const name = `networks/${code}/operations/reports/runs/${run}`;
      runs.set(name, { report: m[1], run, polls: 0 });
      return json(200, { name });
    }

    if ((m = /^(networks\/\d+\/operations\/reports\/runs\/\d+)$/.exec(pathname)) && method === 'GET') {
      const op = runs.get(m[1]);
      if (!op) return notFound();
      op.polls += 1;
      if (op.polls <= completeAfter) return json(200, { name: m[1] });
      return json(200, { name: m[1], done: true, response: { reportResult: `${op.report}/results/${op.run}` } });
    }

    if ((m = /^(networks\/\d+\/reports\/(\d+)\/results\/\d+):fetchRows$/.exec(pathname)) && method === 'GET') {
      const result = m[1];
      const report = reports.get(result.split('/results/')[0]);
      if (!report || expired.has(result)) return notFound();
      const pageSize = Math.min(Number(params.get('pageSize') ?? 1000), 10_000);
      const token = params.get('pageToken');
      if (token !== null && !/^\d+$/.test(token)) {
        return json(400, { error: { code: 400, status: 'INVALID_ARGUMENT', message: 'invalid page token' } });
      }
      const offset = token ? Number(token) : 0;
      const page: Json = { rows: rows.slice(offset, offset + pageSize) };
      if (offset + pageSize < rows.length) page.nextPageToken = String(offset + pageSize);
      if (!token) {
        page.runTime = extra.runTime ?? '2026-09-03T01:00:00Z';
        page.dateRanges = extra.dateRanges ?? [{
          startDate: report.reportDefinition.dateRange?.fixed?.startDate,
          endDate: report.reportDefinition.dateRange?.fixed?.endDate,
        }];
        if (rows.length > 0) page.totalRowCount = rows.length;
      }
      if (page.rows.length === 0) delete page.rows;
      return json(200, page);
    }

    return notFound();
  }

  const fakeFetch = (async (input: RequestInfo | URL, init?: RequestInit) => {
    const url = String(input);
    const method = (init?.method ?? 'GET').toUpperCase();
    const path = url.startsWith(BASE) ? url.slice(BASE.length) : url;
    const body = typeof init?.body === 'string' && init.body.length ? JSON.parse(init.body) : undefined;
    requests.push({ method, path, ...(body !== undefined && { body }) });

    const i = injections.findIndex((x) => x.method === method && x.path.test(path));
    if (i >= 0) {
      const { outcome } = injections.splice(i, 1)[0];
      if (outcome === 'network-error') throw new TypeError('fetch failed');
      if ('succeedThenStatus' in outcome) {
        route(method, path, body);
        return json(outcome.succeedThenStatus, { error: { code: outcome.succeedThenStatus } });
      }
      return json(outcome.status, outcome.body ?? { error: { code: outcome.status } });
    }
    return route(method, path, body);
  }) as typeof fetch;

  return {
    fetch: fakeFetch,
    requests,
    reports,
    network,
    setRows(next, more = {}) { rows = next; extra = more; },
    inject(method, path, outcome) { injections.push({ method, path, outcome }); },
    expire(result) { expired.add(result); },
  };
}
```

```ts
import type { Column, ColumnType } from './values.js';

export type DateSpan = { start: string; end: string };

// What a single report result can say about itself (spec §1.1). Filters and
// sort are copied from the definition GAM ran, in GAM's shapes.
export type Provenance = {
  network_code: string;
  report_type: string;
  report: string;
  result: string;
  run_time: string | null;
  date_range: DateSpan | null;
  date_time_zone: string;
  dimensions: string[];
  metrics: string[];
  filters: unknown[];
  sort: unknown[];
  cache: { hit: boolean };
};

export type InputReport = {
  report: string;
  result: string;
  run_time: string | null;
  date_range: DateSpan | null;
};

// Tools that compute from rows name every report they read (spec §1.4).
export type DerivedProvenance = {
  network_code: string;
  date_time_zone: string;
  inputs: InputReport[];
};

export type ResultEnvelope = {
  summary: string;
  provenance?: Provenance | DerivedProvenance;
  total_row_count: number | null;
  returned_row_count: number;
  truncated: boolean;
  next_cursor: string | null;
  warnings: string[];
  schema: Column[];
  rows: Record<string, unknown>[];
};

// The only place an envelope is assembled. Key order is the contract:
// everything that qualifies the rows is written before them, so a consumer
// that truncates the serialized text loses rows, not caveats.
export function envelope(p: {
  summary: string;
  provenance?: Provenance | DerivedProvenance;
  total_row_count: number | null;
  truncated: boolean;
  next_cursor: string | null;
  warnings: string[];
  schema: Column[];
  rows: Record<string, unknown>[];
}): ResultEnvelope {
  return {
    summary: p.summary,
    ...(p.provenance && { provenance: p.provenance }),
    total_row_count: p.total_row_count,
    returned_row_count: p.rows.length,
    truncated: p.truncated,
    next_cursor: p.next_cursor,
    warnings: p.warnings,
    schema: p.schema,
    rows: p.rows,
  };
}

function typeOf(v: unknown): ColumnType {
  if (typeof v === 'number') return 'number';
  if (typeof v === 'boolean') return 'boolean';
  if (Array.isArray(v)) return 'array';
  return 'string';
}

export function inferSchema(rows: Record<string, unknown>[]): Column[] {
  const first = rows[0];
  if (!first) return [];
  return Object.keys(first).map((name) => ({
    name,
    type: typeOf(rows.find((r) => r[name] !== null && r[name] !== undefined)?.[name]),
  }));
}

// For tools whose rows are not a GAM report (lookups, schema, stats).
export function listEnvelope(
  rows: Record<string, unknown>[],
  summary: string,
  schema?: Column[],
): ResultEnvelope {
  return envelope({
    summary,
    total_row_count: rows.length,
    truncated: false,
    next_cursor: null,
    warnings: [],
    schema: schema ?? inferSchema(rows),
    rows,
  });
}
```

```ts
import type { DateSpan } from './envelope.js';

// A page cursor carries what GAM returns only on a result's first page. It is
// not signed: it names a result the same service account could produce by
// running the report again, so it grants nothing new (spec §1.3).
export type Cursor = {
  v: 1;
  result: string;
  page_token: string;
  offset: number;
  total_row_count: number | null;
  run_time: string | null;
  date_range: DateSpan | null;
};

const BASE64URL = /^[A-Za-z0-9_-]+$/;

export function encodeCursor(c: Omit<Cursor, 'v'>): string {
  return Buffer.from(JSON.stringify({ v: 1, ...c })).toString('base64url');
}

const isCount = (n: unknown) => Number.isInteger(n) && (n as number) >= 0;

export function decodeCursor(text: string, networkCode: string): Cursor | { error: string } {
  if (!BASE64URL.test(text)) return { error: 'cursor is not base64url text' };
  let c: Record<string, unknown>;
  try {
    c = JSON.parse(Buffer.from(text, 'base64url').toString('utf8'));
  } catch {
    return { error: 'cursor does not decode to JSON' };
  }
  if (typeof c !== 'object' || c === null || c.v !== 1) return { error: 'cursor version is not 1' };
  const resultPattern = new RegExp(`^networks/${networkCode}/reports/\\d+/results/\\d+$`);
  if (typeof c.result !== 'string' || !resultPattern.test(c.result)) {
    return { error: `cursor does not name a report result in network ${networkCode}` };
  }
  if (typeof c.page_token !== 'string' || c.page_token.length === 0) return { error: 'cursor has no page token' };
  if (!isCount(c.offset)) return { error: 'cursor offset is not a non-negative integer' };
  if (c.total_row_count !== null && !isCount(c.total_row_count)) return { error: 'cursor row count is invalid' };
  if (c.run_time !== null && typeof c.run_time !== 'string') return { error: 'cursor run time is invalid' };
  const dr = c.date_range as DateSpan | null;
  if (dr !== null && (typeof dr?.start !== 'string' || typeof dr?.end !== 'string')) {
    return { error: 'cursor date range is invalid' };
  }
  return c as unknown as Cursor;
}
```

```ts
import { gamError, type GamErrorEnvelope } from './errors.js';
import { ResultUnavailableError, type DateRange, type ReportQuery, type RowsPage, type StoredReport } from './gam-client.js';
import type { Logger } from './log.js';
import type { ToolContext } from './report-tool-builder.js';
import { reportDefinition, type ReportDefinition } from './report-definition.js';
import { buildTable, type Cell, type Column } from './values.js';
import { encodeCursor, decodeCursor } from './cursor.js';
import { envelope, inferSchema, type DateSpan, type InputReport, type ResultEnvelope } from './envelope.js';

export const GAM_MAX_PAGE_ROWS = 10_000;
export const DEFAULT_PAGE_ROWS = 5_000;
// ponytail: fixed ceiling on rows a computing tool reads; page its output if a real network needs more.
export const MAX_INPUT_ROWS = 50_000;

export type ReportRequest = {
  query: ReportQuery;
  dateRange: DateRange;
  summary: string;
  warnings?: string[];
};

export type ReportRows = {
  columns: Column[];
  rows: Record<string, Cell>[];
  input: InputReport;
  warnings: string[];
};

export function resolvePageRows(raw: string | undefined, logger: Logger): number | { error: string } {
  if (raw === undefined) return DEFAULT_PAGE_ROWS;
  const n = Number(raw);
  if (!Number.isInteger(n) || n < 1) return { error: `GAM_PAGE_ROWS must be a positive integer, got ${raw}` };
  if (n > GAM_MAX_PAGE_ROWS) {
    logger.info('config.page_rows_clamped', { requested: n, used: GAM_MAX_PAGE_ROWS });
    return GAM_MAX_PAGE_ROWS;
  }
  return n;
}

export function pageRows(ctx: ToolContext): number {
  return Math.min(ctx.pageRows ?? DEFAULT_PAGE_ROWS, GAM_MAX_PAGE_ROWS);
}

const realSleep = (ms: number) => new Promise<void>((r) => setTimeout(r, ms));

// The only polling loop in the server.
export async function pollUntilDone(
  ctx: ToolContext,
  operation: string,
): Promise<{ result: string } | GamErrorEnvelope> {
  const timeoutMs = ctx.reportTimeoutMs ?? 300_000;
  const sleep = ctx.sleep ?? realSleep;
  const start = Date.now();
  let delay = 2000;
  for (;;) {
    const s = await ctx.client.pollOperation(operation);
    const elapsed_ms = Date.now() - start;
    ctx.logger.debug('report.poll', { operation, state: s.state, elapsed_ms });
    if (s.state === 'done') {
      ctx.logger.info('report.done', { operation, total_elapsed_ms: elapsed_ms });
      return { result: s.result };
    }
    if (s.state === 'failed') {
      ctx.logger.warn('report.failed', { operation, reason: s.reason });
      return gamError('REPORT_FAILED', { reason: s.reason, operation });
    }
    if (elapsed_ms + delay > timeoutMs) {
      ctx.logger.error('report.timeout', { operation, timeout_ms: timeoutMs });
      return gamError('REPORT_TIMEOUT', { timeout_ms: timeoutMs, operation });
    }
    await sleep(delay);
    delay = Math.min(delay * 1.5, 15_000);
  }
}

async function runToResult(
  ctx: ToolContext,
  query: ReportQuery,
  dateRange: DateRange,
): Promise<{ report: string; result: string } | GamErrorEnvelope> {
  ctx.logger.debug('report.create', {
    dimensions: query.dimensions,
    metrics: query.metrics,
    date_start: dateRange.start,
    date_end: dateRange.end,
  });
  const { report } = await ctx.client.defineReport(query, dateRange);
  const { operation } = await ctx.client.runReport(report);
  const done = await pollUntilDone(ctx, operation);
  return 'error' in done ? done : { report, result: done.result };
}

function toGamError(e: unknown): GamErrorEnvelope {
  if (e instanceof ResultUnavailableError) return gamError('REPORT_RESULT_EXPIRED', { result: e.result });
  return gamError('INTERNAL_ERROR', { reason: e instanceof Error ? e.message : String(e) });
}

// A first page with no totalRowCount and no further page has exactly its own
// rows: proto3 JSON omits zero. With a further page the count is unknown.
function firstPageTotal(page: RowsPage): number | null {
  return page.totalRowCount ?? (page.nextPageToken ? null : page.rows.length);
}

type PageArgs = {
  ctx: ToolContext;
  report: string;
  result: string;
  definition: ReportDefinition;
  summary: string;
  warnings: string[];
  page: RowsPage;
  offset: number;
  total: number | null;
  runTime: string | null;
  dateRange: DateSpan | null;
};

function pageEnvelope(a: PageArgs): ResultEnvelope {
  const table = buildTable(a.definition.dimensions, a.definition.metrics, a.page.rows);
  const returned = table.rows.length;
  const nextCursor = a.page.nextPageToken
    ? encodeCursor({
        result: a.result,
        page_token: a.page.nextPageToken,
        offset: a.offset + returned,
        total_row_count: a.total,
        run_time: a.runTime,
        date_range: a.dateRange,
      })
    : null;
  const warnings = [...a.warnings, ...table.warnings];
  if (a.dateRange === null) warnings.push('GAM did not report the computed date range for this result.');
  if (a.total === null) {
    warnings.push('GAM did not report how many rows this result has.');
  } else if (!nextCursor && a.offset + returned < a.total) {
    warnings.push(`GAM reported ${a.total} rows but returned ${a.offset + returned} and no further page.`);
  }
  return envelope({
    summary: a.summary,
    provenance: {
      network_code: a.ctx.networkCode,
      report_type: a.definition.reportType,
      report: a.report,
      result: a.result,
      run_time: a.runTime,
      date_range: a.dateRange,
      date_time_zone: a.ctx.timezone,
      dimensions: a.definition.dimensions,
      metrics: a.definition.metrics,
      filters: a.definition.filters ?? [],
      sort: [],
      cache: { hit: false },
    },
    total_row_count: a.total,
    truncated: nextCursor !== null || (a.total !== null && a.offset + returned < a.total),
    next_cursor: nextCursor,
    warnings,
    schema: table.columns,
    rows: table.rows,
  });
}

export async function executeReport(
  ctx: ToolContext,
  req: ReportRequest,
): Promise<ResultEnvelope | GamErrorEnvelope> {
  try {
    const ran = await runToResult(ctx, req.query, req.dateRange);
    if ('error' in ran) return ran;
    const page = await ctx.client.fetchRows(ran.result, { pageSize: pageRows(ctx) });
    return pageEnvelope({
      ctx,
      report: ran.report,
      result: ran.result,
      definition: reportDefinition(req.query, req.dateRange),
      summary: req.summary,
      warnings: req.warnings ?? [],
      page,
      offset: 0,
      total: firstPageTotal(page),
      runTime: page.runTime ?? null,
      dateRange: page.dateRanges?.[0] ?? null,
    });
  } catch (e) {
    return toGamError(e);
  }
}

// Stored definitions never change (reports are never patched), so they can be
// remembered per client for the life of the process.
const storedReports = new WeakMap<object, Map<string, StoredReport>>();
const STORED_REPORTS_MAX = 1000;

async function storedReport(ctx: ToolContext, report: string): Promise<StoredReport | null> {
  let byName = storedReports.get(ctx.client);
  if (!byName) storedReports.set(ctx.client, (byName = new Map()));
  const hit = byName.get(report);
  if (hit) return hit;
  const found = await ctx.client.getReport(report);
  if (found) {
    if (byName.size >= STORED_REPORTS_MAX) byName.delete(byName.keys().next().value!);
    byName.set(report, found);
  }
  return found;
}

export async function fetchReportPage(
  ctx: ToolContext,
  cursorText: string,
): Promise<ResultEnvelope | GamErrorEnvelope> {
  const cursor = decodeCursor(cursorText, ctx.networkCode);
  if ('error' in cursor) {
    return gamError('INVALID_INPUT', {
      reason: cursor.error,
      hint: 'Pass next_cursor exactly as a previous result returned it.',
    });
  }
  const report = cursor.result.slice(0, cursor.result.indexOf('/results/'));
  try {
    const stored = await storedReport(ctx, report);
    if (!stored) return gamError('REPORT_RESULT_EXPIRED', { result: cursor.result });
    const page = await ctx.client.fetchRows(cursor.result, { pageSize: pageRows(ctx), pageToken: cursor.page_token });
    const last = cursor.offset + page.rows.length;
    return pageEnvelope({
      ctx,
      report,
      result: cursor.result,
      definition: stored.definition,
      summary: `Rows ${cursor.offset + 1}–${last} of ${cursor.total_row_count ?? 'an unreported number'} from ${report}`,
      warnings: [],
      page,
      offset: cursor.offset,
      total: cursor.total_row_count,
      runTime: cursor.run_time,
      dateRange: cursor.date_range,
    });
  } catch (e) {
    return toGamError(e);
  }
}

function tooLarge(rows: number, limit: number, what: string): GamErrorEnvelope {
  return gamError('RESULT_TOO_LARGE', { rows, limit, what });
}

export async function executeReportRows(
  ctx: ToolContext,
  query: ReportQuery,
  dateRange: DateRange,
): Promise<ReportRows | GamErrorEnvelope> {
  try {
    const ran = await runToResult(ctx, query, dateRange);
    if ('error' in ran) return ran;
    const pages: RowsPage[] = [];
    let token: string | undefined;
    let count = 0;
    do {
      const page = await ctx.client.fetchRows(ran.result, {
        pageSize: GAM_MAX_PAGE_ROWS,
        ...(token && { pageToken: token }),
      });
      if (pages.length === 0 && (page.totalRowCount ?? 0) > MAX_INPUT_ROWS) {
        return tooLarge(page.totalRowCount!, MAX_INPUT_ROWS, 'input report');
      }
      count += page.rows.length;
      if (count > MAX_INPUT_ROWS) return tooLarge(count, MAX_INPUT_ROWS, 'input report');
      pages.push(page);
      token = page.nextPageToken;
    } while (token);
    const table = buildTable(query.dimensions, query.metrics, pages.flatMap((p) => p.rows));
    return {
      columns: table.columns,
      rows: table.rows,
      warnings: table.warnings,
      input: {
        report: ran.report,
        result: ran.result,
        run_time: pages[0].runTime ?? null,
        date_range: pages[0].dateRanges?.[0] ?? null,
      },
    };
  } catch (e) {
    return toGamError(e);
  }
}

export function derivedEnvelope(
  ctx: ToolContext,
  d: { summary: string; inputs: InputReport[]; rows: Record<string, unknown>[]; warnings?: string[] },
): ResultEnvelope | GamErrorEnvelope {
  const limit = pageRows(ctx);
  if (d.rows.length > limit) return tooLarge(d.rows.length, limit, 'output');
  return envelope({
    summary: d.summary,
    provenance: { network_code: ctx.networkCode, date_time_zone: ctx.timezone, inputs: d.inputs },
    total_row_count: d.rows.length,
    truncated: false,
    next_cursor: null,
    warnings: d.warnings ?? [],
    schema: inferSchema(d.rows),
    rows: d.rows,
  });
}
```
